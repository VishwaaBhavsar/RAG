# Output Format Templates

Standardized output formats for the enhance-prompt skill.

## Primary Output Schema

```typescript
interface EnhancedPrompt {
  enhanced: EnhancedSpec;
  decisions: Decision[];
  metadata: ProcessingMetadata;
}

interface EnhancedSpec {
  projectType: ProjectType;
  audience: AudienceSpec;
  complexity: 'simple' | 'medium' | 'complex';
  aestheticDirection: AestheticSpec;
  pageStructure: PageSpec[];
  contentRequirements: ContentSpec;
  technicalContext: TechnicalSpec;
}

type ProjectType =
  | 'landing'
  | 'dashboard'
  | 'ecommerce'
  | 'saas'
  | 'portfolio'
  | 'blog'
  | 'documentation'
  | 'marketplace';

interface AudienceSpec {
  primary: 'b2b' | 'b2c' | 'developer' | 'creative' | 'enterprise';
  demographics: string;
  techSavviness: 'low' | 'medium' | 'high';
  expectations: string[];
}

interface AestheticSpec {
  style: string;           // e.g., "modern-minimalist"
  mood: string;            // e.g., "professional and trustworthy"
  intensity: 'subtle' | 'moderate' | 'bold';
  keywords: string[];      // Extracted aesthetic keywords
  avoidances: string[];    // What NOT to do
}

interface PageSpec {
  name: string;
  route: string;
  sections: SectionSpec[];
  priority: 'critical' | 'important' | 'optional';
  seoRequired: boolean;
}

interface SectionSpec {
  name: string;
  purpose: string;
  components: string[];
  dataNeeds: string[];
}

interface ContentSpec {
  copyTone: string;
  imagery: 'photography' | 'illustrations' | 'abstract' | 'icons' | 'mixed';
  contentDensity: 'sparse' | 'balanced' | 'dense';
  cta: {
    primary: string;
    secondary?: string;
  };
}

interface TechnicalSpec {
  framework: 'next' | 'react' | 'vue';
  responsive: boolean;
  accessibility: 'A' | 'AA' | 'AAA';
  performance: 'standard' | 'optimized' | 'critical';
  features: string[];  // Required technical features
}

interface Decision {
  decision: string;
  reason: string;
  confidence: 'high' | 'medium' | 'low';
  alternatives?: string[];
}

interface ProcessingMetadata {
  originalPrompt: string;
  promptLength: number;
  extractedKeywords: string[];
  processingTime: number;
  version: string;
}
```

  originalPrompt: "Create a modern landing page for my SaaS product"
  promptLength: 48
  extractedKeywords:
    - modern
    - landing page
    - SaaS
  version: "1.0"
```

      route: /dashboard
      priority: critical
      seoRequired: false

    confidence: high

    confidence: medium
```

## Inter-Skill Data Format

When passing to downstream skills, use this compact format:

```yaml
# For brainstorming skill
context:
  type: landing
  audience: b2c
  complexity: medium
  style: modern-minimalist
  pages: [hero, features, social-proof, cta]

# For theme-factory skill
context:
  type: landing
  style: modern-minimalist
  mood: professional
  intensity: moderate

# For frontend-design skill
context:
  type: landing
  pages: [PageSpec...]
  components: [ComponentSpec...]
  style: modern-minimalist
  animations: moderate
```
